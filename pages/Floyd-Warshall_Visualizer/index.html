Floyd-Warshall(G, w):
for i := 1 to n do
for j := 1 to n do
if i == j then dist[i,j,0] := 0
else dist[i,j,0] := ∞
for each (u,v) ∈ E do
dist[u,v,0] := w(u,v)
for k := 1 to n do
for i := 1 to n do
for j := 1 to n do
dist[i,j,k] := min( dist[i,j,k-1],
dist[i,k,k-1] + dist[k,j,k-1] )
return dist[·,·,n]
